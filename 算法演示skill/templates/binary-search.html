    while (left <= right) {
        // 计算中间位置
        int mid = left + (right - left) / 2;

        // 找到目标值
        if (arr[mid] == target) {
            return mid;
        }

        // 目标在左半部分
        else if (arr[mid] > target) {
            right = mid - 1;
        }

        // 目标在右半部分
        else {
            left = mid + 1;
        }
    }

    return -1;  // 未找到
}
